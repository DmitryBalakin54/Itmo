task1

sudo apt install isc-dhcp-server

echo "authoritative;  <--- пишем в файл .conf конфиг сервера
default-lease-time 600;
max-lease-time 7200;

subnet 10.126.207.0 netmask 255.255.255.0 {
    range 10.126.207.10 10.126.207.254;
    option routers 10.126.207.1;   <----- A
    option subnet-mask 255.255.255.0;
    option domain-name-servers 8.8.8.8;
    option broadcast-address 10.126.207.255;
}" | sudo tee /etc/dhcp/dhcpd.conf


sudo ip addr add 10.126.207.1/24 dev tap0 <---- добавляем адресс A

echo 'INTERFACESv4="tap0"' | sudo tee /etc/default/isc-dhcp-server <--- указываем что интерфейс сервера tap0

sudo systemctl restart isc-dhcp-server <--- запуск
sudo systemctl status isc-dhcp-server <--- проверка статуса 

первый чекпоинт готов


sudo apt install radvd <--- ставим для SLAAC

echo "interface tap0 { <--- записываем конфиг
    AdvSendAdvert on;
    MinRtrAdvInterval 3;
    MaxRtrAdvInterval 10;
    prefix fdb0:f2f4:52d8:d8ea::/64 {
        AdvOnLink on;
        AdvAutonomous on;
    };
};" | sudo tee /etc/radvd.conf

sudo sysctl -w net.ipv6.conf.all.forwarding= <--- включаем форвардинг

sudo systemctl restart radvd <--\
sudo systemctl status radvd  <---- стартуем и проверяем статус
