1.G
----------
-- Индекс на PK для Groups, HASH т.к. по id у нас все
-- запросы вида ... WHRE id =: id и нам не нужно группировать
-- по id или искать в диапазоне ... WERE id > N. 
-- ДЗ-5.2.1. Полную информацию о студентах с заданным идентификатором (StudentId, StudentName, GroupName по :StudentId).
-- ДЗ-5.2.2. Полную информацию о студентах с заданным ФИО (StudentId, StudentName, GroupName по :StudentName).
-- ДЗ-5.2.5. Полную информацию о студентах клуба по руководителю (StudentId, StudentName, GroupName по :StudentName).
CREATE UNIQUE INDEX ind_Groups ON Groups USING HASH (GroupId);

-- Индекс для поиска по GroupName, BTREE т.к. нам нужно
-- группировать студентов по группам, а так же покрывающий
-- индекс позволяет не обращаться к таблице, а брать
-- GroupId из индекса.
-- ДЗ-5.2.3. Полную информацию о студентах из заданной группы (StudentId, StudentName, GroupName по :GroupName).
-- ДЗ-7.1.2. Напишите запросы, удаляющие студентов: Учащихся в группе, заданной названием (GroupName).
-- ДЗ-7.3.5. Напишите запросы, обновляющие данные студентов: Перевод всех студентов из группы в группу по названиям (GroupName, FromGroupName).
CREATE INDEX ind_Groups_GroupName ON Groups USING BTREE (GroupName, GroupId);

1.S
---------
-- Индекс на PK для Students, HASH т.к. по id у нас все
-- запросы вида ... WHRE id =: id и нам не нужно группировать
-- по id или искать в диапазоне ... WERE id > N.
-- ДЗ-5.1.1. Информацию о студентах с заданным идентификатором (StudentId, StudentName, GroupId по :StudentId).
-- ДЗ-5.2.1. Полную информацию о студентах с заданным идентификатором (StudentId, StudentName, GroupName по :StudentId).
-- ДЗ-7.3.1. Напишите запросы, обновляющие данные студентов: Изменение имени студента (StudentId, StudentName).
CREATE UNIQUE INDEX ind_Students ON Students USING HASH (StudentId);

-- Индекс для поиска по StudentName, BTREE т.к. можем искать
-- студентов по подстроке из их имени (найти всех Иванов), a
-- так же студенты будут всегда отсортированы, что ускорит запросы,
-- покрывающий индекс позволяет не обращаться к таблице,
-- а брать StudentId из индекса.
-- ДЗ-5.1.2. Информацию о студентах с заданным ФИО (StudentId, StudentName, GroupId по :StudentName).
-- ДЗ-5.2.2. Полную информацию о студентах с заданным ФИО (StudentId, StudentName, GroupName по :StudentName).
-- ДЗ-6.1.1. Информацию о студентах с заданным ФИО (StudentId, StudentName, GroupId по :StudentName).
CREATE INDEX ind_Students_StudentName ON Students USING BTREE (StudentName, StudentId);

-- Индекс на FK таблицы Groups, HASH т.к. нам нужно искать
-- студентов конкретной группы, а так же нам не нужна сортировка.
-- ДЗ-5.2.3. Полную информацию о студентах из заданной группы (StudentId, StudentName, GroupName по :GroupName).
-- ДЗ-6.1.2. Информацию о студентах учащихся в заданной группе (StudentId, StudentName, GroupId по :GroupName).
-- ДЗ-7.3.5. Напишите запросы, обновляющие данные студентов: Перевод всех студентов из группы в группу по названиям (GroupName, FromGroupName).
CREATE INDEX ind_Students_GroupId ON Students USING HASH (GroupId);

2.Q
---------
SELECT AVG(M.Mark) AS AvgMark
FROM Clubs AS CB
NATURAL JOIN ClubMembers AS CM 
NATURAL JOIN Marks AS M
NATURAL JOIN Courses AS C
WHERE CB.ClubName = :ClubName
AND C.CourseName = :CourseName;

FROM Marks AS M
WHERE Mark < 3
GROUP BY StudentId;

---------
